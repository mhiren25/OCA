build_fixture.py now emits the class-name form, and Quantity.kind was already there. Your side is scoring.py, and it's smaller than it looks — the module compares against its own uppercase constants, so mapping once in _state leaves score_field untouched:

python
# Discriminator values are the class names (016), matching Quantity.kind.
# Map them once, here, and refuse anything else.
_STATES = {"Mapped": "MAPPED", "Unmapped": "UNMAPPED", "Unresolved": "UNRESOLVED"}


def _state(node: Any) -> str:
    if node is None:
        return "UNRESOLVED"
    if not isinstance(node, dict):
        raise ValueError(f"not a tri-state node: {node!r}")
    tag = node.get("state")
    if tag not in _STATES:
        raise ValueError(
            f"unrecognised tri-state tag {tag!r}; expected one of "
            f"{', '.join(_STATES)}. A node that reaches scoring without a "
            "readable state is malformed — fix the producer, not this function.")
    return _STATES[tag]
